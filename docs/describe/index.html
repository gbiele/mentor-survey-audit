Survey description

This page describes a cleaned JA-MENTOR youth survey. It uses the same scoring as the local report: Likert sums are prorated when at least 80% of the items are answered, and ACE-IQ follows the World Health Organization category rules.

The workbook stays in this browser. Nothing is uploaded and nothing is stored on the server.

loaded = false
viewof workbook = {
  const input = Inputs.file({
    label: "Cleaned workbook (.xlsx)",
    accept: ".xlsx,.xls",
    multiple: false
  });
  input.id = "workbook-file";
  return input;
}
workbook_name = {
  const chosen = workbook;
  const input = document.querySelector("#workbook-file");
  const file = input && input.files && input.files[0];
  return file ? file.name : (chosen && chosen.name) || "";
}
uploaded = {
  const chosen = workbook;
  const input = document.querySelector("#workbook-file");
  const file = input && input.files && input.files[0];
  if (!file) return false;
  const webR = await webROjs.webRPromise;
  const bytes = new Uint8Array(await file.arrayBuffer());
  await webR.FS.writeFile("upload.xlsx", bytes);
  if (window.XLSX) {
    const book = XLSX.read(bytes, { type: "array" });
    const sheetName = book.SheetNames.includes("Cleaned") ? "Cleaned" : book.SheetNames[0];
    const csv = XLSX.utils.sheet_to_csv(book.Sheets[sheetName]);
    await webR.FS.writeFile("upload.csv", new TextEncoder().encode(csv));
  }
  return true;
}
