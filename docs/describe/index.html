Survey results

This page shows results for a cleaned JA-MENTOR youth survey. It uses the same scoring as the local report: Likert sums are prorated when at least 80% of the items are answered, and ACE-IQ follows the World Health Organization category rules.

Upload the country survey export on the survey auditor. From there, download the cleaned workbook with Export cleaned data, and choose that file here.

The workbook stays in this browser. Nothing is uploaded and nothing is stored on the server.

ImportantTwo steps take a while
  1. Downloading R and the packages. The first time this page is opened, the browser downloads them before a report can be built. Opening the page again in the same browser is quicker.
  2. Producing the plots. After you press Build the report, drawing the charts takes about a minute. The line under the button shows the current step and says when the report is complete.

viewof build_clicks = {
  const file = Inputs.file({
    label: "Cleaned workbook (.xlsx)",
    accept: ".xlsx,.xls",
    multiple: false
  });
  file.id = "workbook-file";
  const button = document.createElement("button");
  button.id = "build-report";
  button.type = "button";
  button.className = "btn btn-primary";
  button.textContent = "Build the report";
  button.style.marginTop = "0.75rem";
  const status = document.createElement("div");
  status.id = "build-status";
  status.setAttribute("data-state", "idle");
  status.setAttribute("role", "status");
  status.setAttribute("aria-live", "polite");
  const spinner = document.createElement("span");
  spinner.className = "spinner-border spinner-border-sm";
  spinner.setAttribute("aria-hidden", "true");
  const message = document.createElement("span");
  message.className = "build-status-text";
  status.append(spinner, message);
  const wrap = document.createElement("div");
  wrap.append(file, button, status);
  wrap.value = 0;
  const fileInput = () => file.querySelector('input[type="file"]');
  const selected = () => {
    const input = fileInput();
    return input && input.files && input.files[0];
  };
  const refreshNote = () => {
    if (status.getAttribute("data-state") === "busy") return;
    status.setAttribute("data-state", "idle");
    const chosen = selected();
    if (!chosen) {
      message.textContent = "Choose a cleaned workbook, then press Build the report.";
    } else {
      message.textContent = chosen.name + " is selected. Press Build the report.";
    }
  };
  file.addEventListener("input", (event) => {
    // Choosing a file only updates the note; it must not count as a click.
    event.stopPropagation();
    refreshNote();
  });
  file.addEventListener("change", refreshNote);
  button.addEventListener("click", () => {
    if (status.getAttribute("data-state") === "busy") return;
    const chosen = selected();
    if (!chosen) {
      refreshNote();
      return;
    }
    // Show the busy state at once; the progress cell takes over from here.
    status.setAttribute("data-state", "busy");
    message.textContent = "Building the report from " + chosen.name + "…";
    button.disabled = true;
    button.textContent = "Building the report…";
    wrap.value = +wrap.value + 1;
    wrap.dispatchEvent(new Event("input", { bubbles: true }));
  });
  refreshNote();
  return wrap;
}

upload_result = {
  const clicks = +build_clicks || 0;
  if (clicks < 1) return { clicks: 0, error: "" };
  const input = document.querySelector('#workbook-file input[type="file"]');
  const file = input && input.files && input.files[0];
  if (!file) return { clicks: 0, error: "" };
  try {
    const webR = await webROjs.webRPromise;
    const bytes = new Uint8Array(await file.arrayBuffer());
    await webR.FS.writeFile("upload.xlsx", bytes);
    if (window.XLSX) {
      const book = XLSX.read(bytes, { type: "array" });
      const sheetName = book.SheetNames.includes("Cleaned") ? "Cleaned" : book.SheetNames[0];
      const csv = XLSX.utils.sheet_to_csv(book.Sheets[sheetName]);
      await webR.FS.writeFile("upload.csv", new TextEncoder().encode(csv));
    }
    return { clicks, error: "" };
  } catch (error) {
    return { clicks, error: error && error.message ? error.message : String(error) };
  }
}

webr_state = {
  const state = { status: "pending", message: "", clicks: 0, failed: {}, last: "", listeners: new Set() };
  const notify = () => state.listeners.forEach((update) => update());
  webROjs.webRPromise.then(
    () => {
      state.status = "ready";
      notify();
    },
    (error) => {
      state.status = "error";
      state.message = error && error.message ? error.message : String(error);
      notify();
    }
  );
  // A section cell that fails in the runtime (worker died) shows an OJS error
  // instead of reporting a status.
  const scan = () => {
    let changed = false;
    document.querySelectorAll(".report-section .exercise-cell").forEach((cell, index) => {
      const broken = cell.matches(".observablehq--error") || cell.querySelector(".observablehq--error");
      if (broken && state.failed[index] !== state.clicks) {
        state.failed[index] = state.clicks;
        changed = true;
      }
    });
    if (changed) notify();
  };
  new MutationObserver(scan).observe(document.body, {
    subtree: true, childList: true, attributes: true, attributeFilter: ["class"]
  });
  return state;
}
build_progress = {
  const first = (value) => Array.isArray(value) ? value[0] : value;
  const clicks = +build_clicks || 0;
  const statuses = [
    section_1, section_2, section_3, section_4, section_5, section_6,
    section_7, section_8, section_9, section_10, section_11, section_12,
    section_13, section_14, section_15, section_16, section_17, section_18
  ];
  // [section id, name, number of cells]; the cells are numbered in page order.
  const layout = [
    ["overview-biography", "Overview: Biography", 1],
    ["overview-socioeconomic-position", "Overview: Socioeconomic position", 1],
    ["overview-bcfpi", "Overview: BCFPI", 2],
    ["overview-ace-iq", "Overview: ACE-IQ", 1],
    ["overview-quality-of-life", "Overview: Quality of life", 1],
    ["overview-resilience", "Overview: Resilience", 1],
    ["overview-social-media-time", "Overview: Social media time", 1],
    ["biography", "Biography", 1],
    ["living-situation", "Living situation", 1],
    ["socioeconomic", "Socioeconomic", 1],
    ["bcfpi-attention", "BCFPI attention", 1],
    ["bcfpi-functioning", "BCFPI functioning", 1],
    ["ace-iq", "ACE-IQ", 2],
    ["kidscreen", "KIDSCREEN", 1],
    ["social-media-time", "Social media time", 1],
    ["social-media-problems", "Social media problems", 1]
  ];
  const state = webr_state;
  state.clicks = clicks;
  const own = (status) => !!status && first(status.build) === clicks;
  const update = () => {
    const status = document.getElementById("build-status");
    const button = document.getElementById("build-report");
    const message = status.querySelector(".build-status-text");
    const classes = ["chart-waiting", "chart-drawn", "chart-failed", "chart-broken", "chart-notdrawn"];
    const paint = (index, kind, text) => {
      const section = document.getElementById(layout[index][0]);
      section.classList.remove(...classes);
      if (kind) section.classList.add("chart-" + kind);
      section.querySelector(".chart-placeholder").textContent = text;
    };
    const paintAll = (kind, text) => layout.forEach((_, index) => paint(index, kind, text));
    const show = (value, text) => {
      const key = clicks + "|" + value + "|" + text;
      if (state.last === key) return;
      state.last = key;
      status.setAttribute("data-state", value);
      message.textContent = text;
      const busy = value === "busy";
      button.disabled = busy;
      button.textContent = busy ? "Building the report…" : "Build the report";
    };
    const waiting = "Waiting for this chart…";
    if (clicks < 1) {
      paintAll(null, "This chart is drawn after you build the report.");
      return;
    }
    if (state.status === "error") {
      paintAll("notdrawn", "Not drawn: R could not be started.");
      show("failed", "R could not be started: " + state.message);
      return;
    }
    if (state.status === "pending") {
      const loading = document.getElementById("exercise-loading-status");
      const step = loading ? loading.textContent.trim() : "";
      paintAll("waiting", waiting);
      show("busy", "Downloading R and the packages" + (step ? " (" + step + ")" : "") + "…");
      return;
    }
    if (!load_status || first(load_status.build) !== clicks) {
      paintAll("waiting", waiting);
      show("busy", "Reading " + (workbook_name || "the workbook") + "…");
      return;
    }
    if (!first(load_status.ok)) {
      paintAll("notdrawn", "Not drawn: the workbook could not be read.");
      show("failed", "The workbook could not be read: " + first(load_status.message));
      return;
    }
    let offset = 0;
    const sections = layout.map(([id, name, count]) => {
      const cells = statuses.slice(offset, offset + count);
      const broken = cells.some((_, i) => state.failed[offset + i] === clicks);
      offset += count;
      const finished = cells.every(own);
      const failed = cells.some((s) => own(s) && !first(s.ok) && !first(s.skipped));
      return { name, finished, failed, broken: broken && !finished };
    });
    // A runtime failure stops the chain: later sections never get a status.
    const stopped = sections.findIndex((s) => s.broken);
    let done = 0;
    sections.forEach((s, index) => {
      if (stopped >= 0 && index > stopped) {
        paint(index, "notdrawn", "Not drawn.");
      } else if (index === stopped) {
        paint(index, "broken", "This section could not be drawn.");
      } else if (s.finished) {
        paint(index, s.failed ? "failed" : "drawn", "");
        done += 1;
      } else {
        paint(index, "waiting", waiting);
      }
    });
    if (stopped >= 0) {
      show("failed", "The report stopped early: " + sections[stopped].name +
        " could not be drawn, so the sections after it were not drawn.");
      return;
    }
    if (done < sections.length) {
      show("busy", "Drawing the charts: " + done + " of " + sections.length + " sections done.");
      return;
    }
    const failedNames = sections.filter((s) => s.failed).map((s) => s.name);
    if (failedNames.length) {
      show("failed", "The report is finished, but these sections could not be drawn: " +
        failedNames.join(", ") + ".");
    } else {
      show("done", "The report is complete.");
    }
  };
  state.listeners.add(update);
  const loading = document.getElementById("exercise-loading-status");
  const observer = new MutationObserver(update);
  if (loading) observer.observe(loading, { subtree: true, childList: true, characterData: true });
  invalidation.then(() => {
    state.listeners.delete(update);
    observer.disconnect();
  });
  update();
}
